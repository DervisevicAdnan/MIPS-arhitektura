##Fibonacci (REKURZIJA TEST) //Nije optimalan treba dugo vremena da se izvrši

main: (0x0)
    addi $a0, $zero, 7       # N=7
    jal fib (0x20)            # F(7)
    j exit (0x60)

fib: (0x20)
    addi $sp, $sp, -8         # 8 byte na stack
    sw $ra, 0($sp)            # spasimo adresu
    sw $a0, 4($sp)            # spasimo n
    slti $t0, $a0, 2          # $t0 uslov. If n < 2 then $t0 = 1, else $t0 = 0
    beq $t0, $zero, L1        # if $t0 == 0  jump L1
    add $v0, $v0, $a0         # $v0 dodajemo n
    addi $sp, $sp, 8          # $sp pokazuje na gornji stek
    jr $ra                    # sljedeca linija koja poziva fibonaccija

L1: (0x40)
    addi $a0, $a0, -1         # n = n - 1
    jal fib                   # F(n - 1)
    addi $a0, $a0, -1         # n = n - 1
    jal fib                   # F(n - 2)
    lw $a0, 4($sp)            # vrati n
    lw $ra, 0($sp)            # brati instrukciju
    addi, $sp, $sp, 8         # vrati pokazivac na stack
    jr $ra                    # vrati na R31
exit: (0x60)
